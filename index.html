/* Banking Planner: Supabase Cloud Sync */ const CLOUD_CONFIG_KEY = "bankPrepSupabaseConfig_v1"; let cloudClient = null; let cloudUser = null; let cloudReady = false; let cloudTimer = null; function cloudStatus(message) { const el = document.getElementById("cloudStatus"); if (el) el.textContent = message; } function configureCloud(projectUrl, publishableKey) { const url = projectUrl.trim().replace(/\/$/, ""); const key = publishableKey.trim(); if (!url.startsWith("https://") || !key) { cloudStatus("Enter a valid Supabase URL and publishable key."); return; } localStorage.setItem( CLOUD_CONFIG_KEY, JSON.stringify({ url, key }) ); cloudClient = window.supabase.createClient(url, key); cloudStatus("Connection saved. Create an account or log in."); } async function cloudSignUp(email, password) { if (!cloudClient) { cloudStatus("Save your Supabase connection settings first."); return; } const { data, error } = await cloudClient.auth.signUp({ email, password }); if (error) { cloudStatus("Signup failed: " + error.message); return; } if (!data.session) { cloudStatus("Check your email to confirm your account, then log in."); return; } cloudUser = data.user; await cloudInitialSync(); } async function cloudLogin(email, password) { if (!cloudClient) { cloudStatus("Save your Supabase connection settings first."); return; } const { data, error } = await cloudClient.auth.signInWithPassword({ email, password }); if (error) { cloudStatus("Login failed: " + error.message); return; } cloudUser = data.user; await cloudInitialSync(); } async function cloudInitialSync() { cloudReady = false; try { const { data, error } = await cloudClient .from("planner_data") .select("data") .eq("user_id", cloudUser.id) .maybeSingle(); if (error) throw error; if (data) { const choice = prompt( "Cloud data exists. Type LOAD to use cloud data, " + "or UPLOAD to replace it with this browser's data." ); if (choice === "LOAD") { state = { ...state, ...data.data }; localStorage.setItem(KEY, JSON.stringify(state)); renderAll(); } else if (choice === "UPLOAD") { await cloudSaveNow(); } else { cloudStatus("Sync cancelled. Local data remains unchanged."); return; } } else { await cloudSaveNow(); } cloudReady = true; cloudStatus("Cloud sync connected."); } catch (error) { cloudStatus("Cloud sync error: " + error.message); } } async function cloudSaveNow() { if (!cloudClient || !cloudUser) return; const { error } = await cloudClient .from("planner_data") .upsert({ user_id: cloudUser.id, data: state, updated_at: new Date().toISOString() }); if (error) throw error; cloudStatus("Cloud saved at " + new Date().toLocaleTimeString()); } function scheduleCloudSave() { if (!cloudReady || !cloudClient || !cloudUser) return; clearTimeout(cloudTimer); cloudTimer = setTimeout(async () => { try { await cloudSaveNow(); } catch (error) { cloudStatus("Cloud save failed: " + error.message); } }, 1200); } async function cloudLogout() { if (cloudClient) await cloudClient.auth.signOut(); cloudUser = null; cloudReady = false; cloudStatus("Logged out. Local saving remains available."); } function loadCloudSettings() { const saved = JSON.parse( localStorage.getItem(CLOUD_CONFIG_KEY) || "null" ); if (!saved || !window.supabase) return; cloudClient = window.supabase.createClient( saved.url, saved.key ); }
